Set Platform Settings (Recommended):

  # Open Rails console
  bundle exec rails console

  # Set SMS from number
  Setting.create!(
    key: 'communications',
    scope_type: 'Platform',
    value: {
      sms: {
        fromNumber: '+17205752095',
        twilioAccountSid: 'ACxxxxx',
        twilioAuthToken: 'xxxxx',
        provider: 'twilio'
      }
    }.to_json
  )

  # Set email from address
  Setting.create!(
    key: 'communications',
    scope_type: 'Platform',
    value: {
      email: {
        fromEmail: 'noreply@yourcompany.com',
        fromName: 'Your Company'
      }
    }.to_json
  )

Or use PlatformSetting (simpler):

  PlatformSetting.set(:sms, :from_number, '+17205752095')
  PlatformSetting.set(:email, :from_email, 'noreply@yourcompany.com')
